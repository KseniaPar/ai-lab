package com.ailab.course.service;

import com.ailab.course.model.Course;
import com.ailab.course.repository.CourseRepository;
import com.ailab.course.repository.LectureRepository;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

@Service
public class SourceSummaryService {

    @Autowired
    private CourseRepository courseRepository;

    @Autowired
    private LectureRepository lectureRepository;

    public SourceSummary getSourceSummary(Long courseId) {
        // Check if the course is owned by the user
        Course course = courseRepository.findById(courseId)
                .orElseThrow(() -> new SecurityException("Course not found or not owned"));

        long audioCount = lectureRepository.countByCourseIdAndSourceType(courseId, "AUDIO");
        long materialCount = lectureRepository.countByCourseIdAndSourceType(courseId, "MATERIAL");
        long readyCount = lectureRepository.countByCourseIdAndReadyStatus(courseId, true);

        return new SourceSummary(
                course,
                lectureRepository.findByCourseId(courseId),
                audioCount,
                materialCount,
                readyCount
        );
    }
}
